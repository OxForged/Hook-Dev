Integrate in one interface

A Latch implements a single interface and registers against a pool. The registry enforces permissions, gas caps and lifecycle order, so the host protocol never has to trust your bytecode.

— beforeSwap / afterSwap callbacks
— deterministic latch addresses
— revert isolation per hook
— on-chain permission bitmap
FeeLatch.sol
// dynamic fee latch
import {ILatch} from "@latch/core/ILatch.sol";
contract FeeLatch is ILatch {
function permissions() external pure returns (uint16) {
return BEFORE_SWAP | AFTER_SWAP;
}
function beforeSwap(PoolKey calldata key, SwapParams calldata p)
external override returns (uint24 fee)
{
fee = volatility(key) > THRESHOLD ? 3000 : 500;
}
}
